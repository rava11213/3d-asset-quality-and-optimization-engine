cmake_minimum_required(VERSION 3.16)
project(assetq VERSION 0.1.0 LANGUAGES CXX)
add_library(assetq_mesh cpp/src/mesh.cpp cpp/src/simplify.cpp)
target_include_directories(assetq_mesh PUBLIC cpp/include)
target_compile_features(assetq_mesh PUBLIC cxx_std_17)
add_executable(assetq-inspect cpp/src/main.cpp)
target_link_libraries(assetq-inspect PRIVATE assetq_mesh)
add_executable(assetq-simplify cpp/src/simplify_main.cpp)
target_link_libraries(assetq-simplify PRIVATE assetq_mesh)
if(MSVC)
  target_compile_options(assetq_mesh PRIVATE /W4)
else()
  target_compile_options(assetq_mesh PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS assetq-inspect assetq-simplify RUNTIME DESTINATION bin)
